bool initializHardware(){
  Wire.begin(IIC_SDA, IIC_SCL);
  if (!rtcSensor.init(Wire, PCF85063_SLAVE_ADDRESS, IIC_SDA, IIC_SCL)) {
    USBSerial.println("RTC sensor init failed!");
    return false;
  }
  return true;

}

bool isWifiAP = false;

void loop() {
  isWifiAP = switchWiFiAP();
  if (wifiAP != isWifiAP)
  {
    wifiAP = isWifiAP;
    if (wifiAP){
      comm.init();
      control.init();
      static Webapp web; // local static so it persists
      web.init();
    }
    else{
      comm.stop();
    }
  }

  static uint32_t lastCheck = 0;
  if (millis() - lastCheck > 10000){
    lastCheck = millis();
    HH_MM = RTC_TIME.printTime();
  }

  comm.loop();
  control.loop();

  setTimeHHMM((HH_MM).c_str());

  lvgl_handler();
  ui_tick();

  delay(5);
}

void applyWiFiMode(void *param){
  bool wifiAP = false;
  while (true)
  {
    if (wifiAP != isWifiAP)
    {
      wifiAP = isWifiAP;
      if (wifiAP){
        comm.init();
        control.init();
        static Webapp web; // local static so it persists
        web.init();
      }
      else{
        comm.stop();
      }
    }

    static uint32_t lastCheck = 0;
    if (millis() - lastCheck > 10000){
      lastCheck = millis();
      HH_MM = RTC_TIME.printTime();
    }

    comm.loop();
    control.loop();

    vTaskDelay(10);
    
  }
  
}

void applyLVGLTheme(void *param){
  lvgl_init();
  USBSerial.println("LVGL initialized");
  ui_init();
  USBSerial.println("UI initialized");
  while (true)
  {
    isWifiAP = switchWiFiAP();
    setTimeHHMM((HH_MM).c_str());
    lvgl_handler();
    ui_tick();
    vTaskDelay(10);
  }
  
}
